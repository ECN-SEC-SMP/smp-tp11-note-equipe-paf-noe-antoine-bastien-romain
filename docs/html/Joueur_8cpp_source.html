47std::vector<CarteTrain> Joueur::selectionnerCartes(Couleur couleur, int longueur) const {
48 std::vector<CarteTrain> selection;
49 std::vector<CarteTrain> locomotives;
50 int nbSelectionnes = 0;
51
52 for (const auto& c : mainCartes) {
53 if (c.estLocomotive()) {
54 locomotives.push_back(c);
55 } else if (c.getCouleur() == couleur && nbSelectionnes < longueur) {
56 selection.push_back(c);
57 nbSelectionnes++;
58 }
59 }
60
61 int manquantes = longueur - nbSelectionnes;
62 for (const auto& loco : locomotives) {
63 if (manquantes <= 0) break;
64 selection.push_back(loco);
65 manquantes--;
66 nbSelectionnes++;
67 }
68
69 if (nbSelectionnes < longueur) {
